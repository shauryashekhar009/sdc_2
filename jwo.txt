pipeline {
    agent any

    tools {
        maven 'Maven-home'
    }

    stages {
        stage('Git Repo & Clean') {
            steps {
                git branch: 'master', url: 'https://github.com/shauryashekhar009/hello-world-war.git'
                sh 'mvn clean'
            }
        }
        stage('Install') {
            steps {
                sh 'mvn install'
            }
        }
        stage('Test') {
            steps {
                sh 'mvn test'
            }
        }
        stage('Package') {
            steps {
                sh 'mvn package'
            }
        }
        stage('Deploy to Tomcat') {
            steps {
                writeFile file: 'Dockerfile', text: '''FROM tomcat:9.0
COPY target/*.war /usr/local/tomcat/webapps/ROOT.war
'''
                sh 'docker build -t hello-war-tomcat .'
                sh 'docker rm -f hello-war-app || true'
                sh 'docker run -d --name hello-war-app -p 9090:8080 hello-war-tomcat'
            }
        }
    }

    post {
        success {
            archiveArtifacts artifacts: 'target/*.war'
            echo 'App deployed! Open http://localhost:9090'
        }
    }
}
