pipeline {
    agent any

    tools {
        maven 'Maven-home'
    }

    stages {
        stage('Git Repo & Clean') {
            steps {
                git branch: 'master', url: 'https://github.com/shauryashekhar009/hello-world-war.git'
                sh 'mvn clean'
            }
        }
        stage('Install') {
            steps {
                sh 'mvn install'
            }
        }
        stage('Test') {
            steps {
                sh 'mvn test'
            }
        }
        stage('Package') {
            steps {
                sh 'mvn package'
            }
        }
    }

    post {
        success {
            archiveArtifacts artifacts: 'target/*.war'
            echo 'All 4 stages passed!'
        }
    }
}
